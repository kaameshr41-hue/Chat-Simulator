<script>
/* ====== YOUR CUSTOM MESSAGES (final secret is hidden until tapped) ====== */
const messages = [
  {who:'them', text: "Hey you know something"},
  {who:'them', text: "wait wait don't think too long"},
  {who:'them', text: "you are the sweetest person I have ever met"},
  // final secret — popup: true hides the real text in the bubble and reveals only on tap
  {who:'them', text: "Aadthaa😂", popup: true}
];

const chatEl = document.getElementById('chat');
const overlay = document.getElementById('overlay');
const modalMsg = document.getElementById('modalMsg');
const closeModalBtn = document.getElementById('closeModal');
const copyBtn = document.getElementById('copyMsg');

/* create typing bubble */
function appendTypingBubble(side){
  const row = document.createElement('div'); row.className = 'row ' + (side==='me'?'me':'');
  const wrap = document.createElement('div'); wrap.className = 'bubble typing';
  wrap.setAttribute('aria-hidden','true');
  wrap.innerHTML = `<span style="display:inline-flex;align-items:center"><span class="dot"></span><span class="dot"></span><span class="dot"></span></span>`;
  row.appendChild(wrap);
  chatEl.appendChild(row);
  chatEl.scrollTop = chatEl.scrollHeight;
  return row;
}

/* append normal message bubble */
function appendMessageBubble(who, text, time, isPopup){
  const row = document.createElement('div'); row.className = 'row ' + (who==='me'?'me':'');
  const bubble = document.createElement('div'); bubble.className = 'bubble ' + (who==='me'?'me':'them');
  bubble.setAttribute('tabindex','0');
  bubble.setAttribute('role','button');
  // store real text in dataset (hidden if popup)
  bubble.dataset.text = text;
  bubble.dataset.popup = isPopup ? '1' : '0';

  // If this message is a popup secret, show a masked hint instead of the real text
  if (isPopup) {
    bubble.innerHTML = `<div class="content">Tap to reveal 🔒</div>`;
    // small accessibility hint
    bubble.setAttribute('aria-label', 'Secret message. Tap to reveal.');
    // reveal only via click/keyboard
    bubble.addEventListener('click', ()=> openPreview(bubble.dataset.text));
    bubble.addEventListener('keydown', (e)=> { if(e.key==='Enter' || e.key===' ') openPreview(bubble.dataset.text) });
  } else {
    bubble.innerHTML = `<div class="content">${escapeHtml(text)}</div>`;
    bubble.addEventListener('click', ()=> {
      // small press feedback
      bubble.style.transition = 'transform 140ms';
      bubble.style.transform = 'scale(.995)';
      setTimeout(()=> bubble.style.transform = '', 140);
    });
    bubble.addEventListener('keydown', (e)=> { if(e.key==='Enter' || e.key===' ') {
      bubble.style.transform = 'scale(.995)';
      setTimeout(()=> bubble.style.transform = '', 140);
    }});
  }

  const meta = document.createElement('div'); meta.className = 'meta';
  const ts = document.createElement('span'); ts.textContent = time || currentTime();
  meta.appendChild(ts);
  bubble.appendChild(meta);
  row.appendChild(bubble);
  chatEl.appendChild(row);
  requestAnimationFrame(()=> bubble.classList.add('show'));
  chatEl.scrollTop = chatEl.scrollHeight;
}

/* current local time HH:MM */
function currentTime(){
  const d = new Date();
  const hh = String(d.getHours()).padStart(2,'0');
  const mm = String(d.getMinutes()).padStart(2,'0');
  return hh + ':' + mm;
}

function escapeHtml(s){
  return s.replaceAll('&','&amp;').replaceAll('<','&lt;').replaceAll('>','&gt;');
}

/* play conversation with typing durations */
async function playConversation(){
  for(let i=0;i<messages.length;i++){
    const msg = messages[i];
    const typingRow = appendTypingBubble(msg.who==='me'?'me':'them');
    const base = 420;
    const perChar = 45;
    const est = msg.delay ? msg.delay : Math.min(base + msg.text.length * perChar, 3800);
    await wait(est);
    typingRow.remove();
    appendMessageBubble(msg.who, msg.text, msg.time, !!msg.popup);
    await wait(220);
  }
}

/* wait helper */
function wait(ms){ return new Promise(res=> setTimeout(res, ms)); }

/* modal functions */
function openPreview(text){
  modalMsg.textContent = ''; // clear
  modalMsg.textContent = text;
  overlay.classList.add('show');
  overlay.setAttribute('aria-hidden','false');
  closeModalBtn.focus();
  if (navigator.vibrate) try { navigator.vibrate(40); } catch(e) {}
}
function closePreview(){
  overlay.classList.remove('show');
  overlay.setAttribute('aria-hidden','true');
}

/* copy modal text */
async function copyToClipboard(){
  const txt = modalMsg.textContent;
  try{
    await navigator.clipboard.writeText(txt);
    copyBtn.textContent = 'Copied!';
    setTimeout(()=> copyBtn.textContent = 'Copy', 1200);
  }catch(e){
    copyBtn.textContent = 'Failed';
    setTimeout(()=> copyBtn.textContent = 'Copy', 1200);
  }
}

/* modal controls */
closeModalBtn.addEventListener('click', closePreview);
copyBtn.addEventListener('click', copyToClipboard);
overlay.addEventListener('click', (e)=> { if(e.target===overlay) closePreview() });
document.addEventListener('keydown', (e)=> { if(e.key==='Escape' && overlay.classList.contains('show')) closePreview() });

/* start conversation */
playConversation();

/* URL custom params optional */
(function applyUrlCustom(){
  const params = new URLSearchParams(location.search);
  const name = params.get('name');
  const avatar = params.get('avatar');
  const status = params.get('status');
  if(name) document.getElementById('contactName').textContent = name;
  if(avatar) document.getElementById('avatar').textContent = avatar;
  if(status) document.getElementById('contactStatus').textContent = status;
})();
